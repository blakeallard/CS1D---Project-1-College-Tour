164void selectHelper(QueryResult &results, Statement &st)
165{
166 while (st.step())
167 {
168 Row row;
169 int colCount = sqlite3_column_count(st.getStatement());
170
171 for (int i = 0; i < colCount; ++i)
172 {
173 std::string name = sqlite3_column_name(st.getStatement(), i);
174 int type = sqlite3_column_type(st.getStatement(), i);
175
176 switch (type)
177 {
178 case SQLITE_INTEGER:
179 row[name] = sqlite3_column_int(st.getStatement(), i);
180 break;
181
182 case SQLITE_FLOAT:
183 row[name] = sqlite3_column_double(st.getStatement(), i);
184 break;
185
186 case SQLITE_TEXT:
187 row[name] = std::string(reinterpret_cast<const char *>(
188 sqlite3_column_text(st.getStatement(), i)));
189 break;
190
191 case SQLITE_NULL:
192 row[name] = nullptr;
193 break;
194 }
195 }
196
197 results.push_back(row);
198 }
199}
