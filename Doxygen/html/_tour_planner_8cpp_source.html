28void TourPlanner::getAllDistances(const vector<string> campusNames,
29 vector<vector<double>> &matrix,
30 map<string, int> &campusIndex)
31{
32 // Query distances for each campus - O(n) queries
33 for (int i = 0; i < campusNames.size(); i++)
34 {
35 // Database query for all distances from this campus
36 QueryData::QueryResult allDistances = QueryData::selectRows(
37 "distances.db", "distances",
38 {"starting_college", "ending_college", "distance"},
39 {"starting_college"}, {campusNames[i]});
40
41 // Process each row from the query result
42 for (auto &row : allDistances)
43 {
44 string start_college = get<string>(row.at("starting_college"));
45 string end_college = get<string>(row.at("ending_college"));
46 double distance = get<double>(row.at("distance"));
47
48 // Only add if both campuses are in our tour set
49 if (campusIndex.find(start_college) != campusIndex.end() &&
50 campusIndex.find(end_college) != campusIndex.end())
51 {
52 int i = campusIndex[start_college];
53 int j = campusIndex[end_college];
54
55 // Fill both directions (symmetric matrix)
56 matrix[i][j] = distance;
57 matrix[j][i] = distance;
58 }
59 }
60 }
61}

74void TourPlanner::visit(const int current,
75 const vector<vector<double>> &distances,
76 vector<bool> &visited, vector<int> &route)
77{
78 // Add current campus to route and mark visited
79 route.push_back(current);
80 visited[current] = true;
81
82 // Build min-heap of distances to unvisited campuses
83 // LinkedHeapTree provides O(log n) insert and O(log n) removeMin
84 LinkedHeapTree<double, int> heap;
85
86 // Insert all unvisited campuses with their distances - O(n log n)
87 for (int i = 0; i < visited.size(); i++)
88 {
89 if (!visited[i])
90 heap.insert(distances[current][i], i);
91 }
92
93 // Base case: all campuses visited
94 if (heap.empty())
95 return;
96
97 // Greedy choice: select nearest unvisited campus - O(log n)
98 int next = heap.removeMin();
99
100 // Recursive call to visit next campus
101 visit(next, distances, visited, route);
102}

113void TourPlanner::twoOpt(vector<int> &route, const vector<vector<double>> &dist)
114{
115 bool improved = true;
116
117 // Continue until no improvement found
118 while (improved)
119 {
120 improved = false;
121
122 // Check all pairs of non-adjacent edges - O(n²)
123 for (int i = 1; i < route.size() - 2; i++)
124 {
125 for (int j = i + 1; j < route.size() - 1; j++)
126 {
127 // Current edges: A-B and C-D
128 int A = route[i - 1];
129 int B = route[i];
130 int C = route[j];
131 int D = route[j + 1];
132
133 // Current total distance of these two edges
134 double current = dist[A][B] + dist[C][D];
135
136 // Proposed distance if we reverse segment B...C
137 // New edges would be: A-C and B-D
138 double proposed = dist[A][C] + dist[B][D];
139
140 // If proposed is better, reverse the segment
141 if (proposed < current)
142 {
143 // Reverse segment from B to C (inclusive)
144 reverse(route.begin() + i, route.begin() + j + 1);
145
146 improved = true;
147 }
148 }
149 }
150 }
151}

166TourResult TourPlanner::calculateOptimalTour(
167 const std::string &startCampus,
168 const std::vector<std::string> &campusesToVisit)
169{
170 // Build complete campus list with start campus first
171 vector<string> campuses;
172 campuses.push_back(startCampus);
173 for (int i = 0; i < campusesToVisit.size(); i++)
174 {
175 campuses.push_back(campusesToVisit[i]);
176 }
177 int N = campuses.size();
178
179 // Create index mapping: campus name -> matrix index
180 // Using map for O(log n) lookup
181 map<string, int> campusIndex;
182 int nextIndex = 0;
183 for (int i = 0; i < N; i++)
184 {
185 campusIndex[campuses[i]] = nextIndex;
186 nextIndex++;
187 }
188
189 // Initialize data structures
190 vector<int> route; // Final route (indices)
191 vector<bool> visited(N, 0); // Visited tracking
192 vector<vector<double>> allDistancesMatrix(N, vector<double>(N, 0)); // Distance matrix
193
194 // Populate distance matrix from database - O(n)
195 getAllDistances(campuses, allDistancesMatrix, campusIndex);
196
197 // Build initial route using greedy nearest-neighbor - O(n²)
198 visit(0, allDistancesMatrix, visited, route);
199
200 // Optimize route with 2-opt local search - O(n²) per iteration
201 twoOpt(route, allDistancesMatrix);
202
203 // Convert route indices to TourResult format
204 double total = 0;
205 TourResult results;
206
207 // Add starting campus with 0 distance
208 TourStop stop1;
209 stop1.campus = startCampus;
210 stop1.distanceFromPrevious = 0;
211 results.stops.push_back(stop1);
212
213 // Add remaining stops with distances
214 for (int i = 1; i < route.size(); i++)
215 {
216 TourStop stop;
217 stop.campus = campuses[route[i]];
218 stop.distanceFromPrevious = allDistancesMatrix[route[i]][route[i - 1]];
219 total += allDistancesMatrix[route[i]][route[i - 1]];
220 results.stops.push_back(stop);
221 }
222 results.totalDistance = total;
223
224 return results;
225}
