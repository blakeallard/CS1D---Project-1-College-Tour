159 {

172 {
173 auto jsonData = crow::json::load(req.body);
174
175 if (!jsonData)
176 {
177 return crow::response(400, "Invalid JSON");
178 }
179
180 string startCampus = jsonData["startCampus"].s();
181 int numberOfCampuses = jsonData["numberOfCampuses"].i();
182
183 // Get all campuses reachable from startCampus with distances
184 QueryData::QueryResult allDistances = QueryData::selectRows(
185 "distances.db", "distances",
186 {"ending_college", "distance"},
187 {"starting_college"}, {startCampus});
188
189 // Sort by distance and select N-1 nearest
190 vector<pair<string, double>> campusDistances;
191 for (const auto &row : allDistances)
192 {
193 string endCampus = get<string>(row.at("ending_college"));
194 double dist = get<double>(row.at("distance"));
195 campusDistances.push_back({endCampus, dist});
196 }
197
198 // Sort by distance ascending - O(n log n)
199 sort(campusDistances.begin(), campusDistances.end(),
200 [](const pair<string, double> &a, const pair<string, double> &b) {
201 return a.second < b.second;
202 });
203
204 // Select the N-1 nearest campuses (N includes the starting campus)
205 vector<string> selectedCampuses;
206 int campusesToSelect = min(numberOfCampuses - 1, (int)campusDistances.size());
207 for (int i = 0; i < campusesToSelect; i++)
208 {
209 selectedCampuses.push_back(campusDistances[i].first);
210 }
211
212 // Calculate optimal tour through selected campuses - O(n^2)
213 TourResult tour = TourPlanner::calculateOptimalTour(startCampus, selectedCampuses);
214
